### 1. LLM Factory（三层抽象）

**文件**: `src/llm/factory.py`

```python
from enum import Enum
from typing import Optional
from langchain_anthropic import ChatAnthropic
from langchain_openai import ChatOpenAI
from tenacity import retry, stop_after_attempt, wait_exponential

class LLMTier(str, Enum):
    HIGH = "high"
    MEDIUM = "medium"
    LIGHT = "light"

class LLMFactory:
    """LLM 三层抽象工厂"""

    _config = {
        "high": {
            "provider": "anthropic",
            "model": "claude-opus-4-6",
            "temperature": 0.1,
            "max_tokens": 4096
        },
        "medium": {
            "provider": "anthropic",
            "model": "claude-sonnet-4-6",
            "temperature": 0.3,
            "max_tokens": 4096
        },
        "light": {
            "provider": "ollama",
            "model": "qwen3.5:32b",
            "temperature": 0.5,
            "max_tokens": 2048
        }
    }

    @classmethod
    @retry(stop=stop_after_attempt(3), wait=wait_exponential(min=1, max=10))
    def get_llm(cls, tier: LLMTier, fallback: bool = True):
        """
        获取指定层级的 LLM

        Args:
            tier: 层级（high/medium/light）
            fallback: 失败时是否降级

        Returns:
            LLM 实例
        """
        try:
            config = cls._config[tier]
            provider = config["provider"]

            if provider == "anthropic":
                return ChatAnthropic(
                    model=config["model"],
                    temperature=config["temperature"],
                    max_tokens=config["max_tokens"]
                )
            elif provider == "openai":
                return ChatOpenAI(
                    model=config["model"],
                    temperature=config["temperature"],
                    max_tokens=config["max_tokens"]
                )
            elif provider == "ollama":
                from langchain_community.llms import Ollama
                return Ollama(
                    model=config["model"],
                    temperature=config["temperature"]
                )
        except Exception as e:
            if fallback and tier != LLMTier.LIGHT:
                # 降级策略：high -> medium -> light
                next_tier = LLMTier.MEDIUM if tier == LLMTier.HIGH else LLMTier.LIGHT
                logger.warning(f"{tier} failed, falling back to {next_tier}")
                return cls.get_llm(next_tier, fallback=False)
            raise
```

### 2. 混合检索模块

**文件**: `src/retrieval/hybrid_search.py`

```python
from FlagEmbedding import BGEM3FlagModel
from chromadb import PersistentClient
from typing import List, Dict

class HybridSearcher:
    """基于 BGE-M3 的混合检索"""

    def __init__(self, chroma_path: str = "./data/chroma_db"):
        # 初始化 BGE-M3（支持 dense + sparse）
        self.model = BGEM3FlagModel(
            'BAAI/bge-m3',
            use_fp16=False  # CPU 模式
        )

        # 初始化 Chroma
        self.client = PersistentClient(path=chroma_path)
        self.collection = self.client.get_or_create_collection(
            name="documents",
            metadata={"hnsw:space": "cosine"}
        )

    def search(
        self,
        query: str,
        top_k: int = 10,
        filters: Optional[Dict] = None
    ) -> List[Dict]:
        """
        混合检索

        Args:
            query: 查询文本
            top_k: 返回结果数
            filters: 元数据过滤条件

        Returns:
            检索结果列表
        """
        # 1. 生成混合向量
        embeddings = self.model.encode(
            [query],
            return_dense=True,
            return_sparse=True
        )

        # 2. Dense 检索
        dense_results = self.collection.query(
            query_embeddings=embeddings['dense_vecs'].tolist(),
            n_results=top_k,
            where=filters
        )

        # 3. Sparse 检索（BM25-like）
        sparse_results = self._sparse_search(
            embeddings['lexical_weights'][0],
            top_k=top_k,
            filters=filters
        )

        # 4. 混合排序（RRF - Reciprocal Rank Fusion）
        final_results = self._rrf_fusion(
            dense_results,
            sparse_results,
            k=60
        )

        return final_results[:top_k]

    def _rrf_fusion(self, dense_results, sparse_results, k=60):
        """RRF 融合算法"""
        scores = {}

        # Dense 结果打分
        for rank, doc_id in enumerate(dense_results['ids'][0]):
            scores[doc_id] = scores.get(doc_id, 0) + 1 / (k + rank + 1)

        # Sparse 结果打分
        for rank, doc_id in enumerate(sparse_results['ids']):
            scores[doc_id] = scores.get(doc_id, 0) + 1 / (k + rank + 1)

        # 按分数排序
        sorted_docs = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return sorted_docs
```

### 3. 文档摄入模块

**文件**: `src/ingestion/converter.py`

```python
from docling.document_converter import DocumentConverter
from typing import List, Dict
import logging

class DocumentIngestion:
    """文档摄入与处理"""

    def __init__(self, batch_size: int = 20):
        self.converter = DocumentConverter()
        self.batch_size = batch_size
        self.logger = logging.getLogger(__name__)

    def process_documents(
        self,
        file_paths: List[str],
        chunk_size: int = 512,
        chunk_overlap: int = 50
    ) -> List[Dict]:
        """
        批量处理文档

        Args:
            file_paths: 文档路径列表
            chunk_size: 分块大小
            chunk_overlap: 重叠大小

        Returns:
            处理后的文档块列表
        """
        all_chunks = []

        # 分批处理（适配低内存）
        for i in range(0, len(file_paths), self.batch_size):
            batch = file_paths[i:i + self.batch_size]
            self.logger.info(f"Processing batch {i//self.batch_size + 1}")

            for file_path in batch:
                try:
                    # 1. 转换文档
                    result = self.converter.convert(file_path)

                    # 2. 提取元数据
                    metadata = self._extract_metadata(result)

                    # 3. 智能分块
                    chunks = self._chunk_document(
                        result.document.export_to_markdown(),
                        chunk_size=chunk_size,
                        overlap=chunk_overlap,
                        metadata=metadata
                    )

                    all_chunks.extend(chunks)

                except Exception as e:
                    self.logger.error(f"Failed to process {file_path}: {e}")
                    continue

        return all_chunks

    def _chunk_document(
        self,
        text: str,
        chunk_size: int,
        overlap: int,
        metadata: Dict
    ) -> List[Dict]:
        """智能分块（保持语义完整性）"""
        # 按段落分割
        paragraphs = text.split('\n\n')
        chunks = []
        current_chunk = ""

        for para in paragraphs:
            if len(current_chunk) + len(para) < chunk_size:
                current_chunk += para + "\n\n"
            else:
                if current_chunk:
                    chunks.append({
                        "text": current_chunk.strip(),
                        "metadata": metadata
                    })
                current_chunk = para + "\n\n"

        if current_chunk:
            chunks.append({
                "text": current_chunk.strip(),
                "metadata": metadata
            })

        return chunks
```

from src.agents.base_agent import BaseAgent
from src.llm.factory import LLMTier

class PlanningAgent(BaseAgent):

        super().__init__(tier=LLMTier.HIGH)

### 成本追踪实现

**文件**: `src/llm/cost_tracker.py`

```python
from typing import Dict
from datetime import datetime
import json

class CostTracker:
    """成本追踪器"""

    # 价格表（每 1M tokens）
    PRICING = {
        "claude-opus-4-6": {"input": 15.0, "output": 75.0},
        "claude-sonnet-4-6": {"input": 3.0, "output": 15.0},
        "claude-haiku-4-5": {"input": 0.8, "output": 4.0},
        "gpt-4o": {"input": 2.5, "output": 10.0},
        "gpt-4o-mini": {"input": 0.15, "output": 0.6},
    }

    _records = []

    @classmethod
    def track(cls, agent: str, tier: str, model: str,
              input_tokens: int, output_tokens: int):
        """记录一次调用"""
        pricing = cls.PRICING.get(model, {"input": 0, "output": 0})
        cost = (
            input_tokens / 1_000_000 * pricing["input"] +
            output_tokens / 1_000_000 * pricing["output"]
        )

        record = {
            "timestamp": datetime.now().isoformat(),
            "agent": agent,
            "tier": tier,
            "model": model,
            "input_tokens": input_tokens,
            "output_tokens": output_tokens,
            "cost": cost
        }

        cls._records.append(record)

    @classmethod
    def get_report(cls) -> Dict:
        """生成成本报告"""
        total_cost = sum(r["cost"] for r in cls._records)
        by_tier = {}

        for record in cls._records:
            tier = record["tier"]
            by_tier[tier] = by_tier.get(tier, 0) + record["cost"]

        return {
            "total_cost": total_cost,
            "by_tier": by_tier,
            "total_calls": len(cls._records)
        }
```

## 数据流设计

### Agentic RAG 工作流（LangGraph）

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict

class AgentState(TypedDict):
    requirement: str
    doc_type: str
    outline: Dict
    retrieved_docs: List[Dict]
    reasoning_result: Dict
    content: str
    review_result: Dict
    iteration_count: int

def create_workflow():
    """创建 Agentic RAG 工作流"""

    workflow = StateGraph(AgentState)

    # 添加节点
    workflow.add_node("planning", planning_agent.run)
    workflow.add_node("retrieval", retrieval_agent.run)
    workflow.add_node("reasoning", reasoning_agent.run)
    workflow.add_node("writing", writing_agent.run)
    workflow.add_node("review", review_agent.run)

    # 定义边
    workflow.set_entry_point("planning")
    workflow.add_edge("planning", "retrieval")
    workflow.add_edge("retrieval", "reasoning")

    # 条件边：是否需要继续检索
    workflow.add_conditional_edges(
        "reasoning",
        should_continue_retrieval,
        {
            "continue": "retrieval",  # 信息不足，继续检索
            "proceed": "writing"      # 信息充分，开始撰写
        }
    )

    workflow.add_edge("writing", "review")
    workflow.add_edge("review", END)

    return workflow.compile()

def should_continue_retrieval(state: AgentState) -> str:
    """判断是否继续检索"""
    if state["iteration_count"] >= 5:
        return "proceed"  # 最多 5 轮

    if state["reasoning_result"]["is_sufficient"]:
        return "proceed"
    else:
        return "continue"
```

### 1. 内存优化

- **分批处理**：文档摄入每 20 篇一批

- **缓存策略**：
  - 嵌入向量缓存
  - 检索结果缓存（TTL 1 小时）
  - LLM 响应缓存（相似查询）

- **三层路由**：高频低价值任务用 Light 层
- **批量调用**：合并多个小请求
- **提示词优化**：减少不必要的 token

from pydantic_settings import BaseSettings

class Settings(BaseSettings):
    # LLM 配置
    llm_high_provider: str = "anthropic"
    llm_high_model: str = "claude-opus-4-6"
    llm_medium_provider: str = "anthropic"
    llm_medium_model: str = "claude-sonnet-4-6"
    llm_light_provider: str = "ollama"
    llm_light_model: str = "qwen3.5:32b"

    # API Keys
    anthropic_api_key: str
    openai_api_key: str = ""

    # 存储配置
    chroma_path: str = "./data/chroma_db"

    # 性能配置
    batch_size: int = 20
    chunk_size: int = 512
    chunk_overlap: int = 50
    max_retrieval_iterations: int = 5

    class Config:
        env_file = ".env"
        env_file_encoding = "utf-8"

settings = Settings()

from fastapi import FastAPI, BackgroundTasks

app = FastAPI()
